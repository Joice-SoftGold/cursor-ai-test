import { useState } from 'react'

function DashboardComponent() {
  const [count, setCount] = useState(0)
  const [isActive, setIsActive] = useState(false)

  return (
    <div className="min-h-screen bg-gradient-to-br from-purple-500 via-pink-500 to-red-500 flex items-center justify-center p-4">
      <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-md w-full">
        <h1 className="text-4xl font-bold text-gray-800 mb-4 text-center">
          Welcome to My App
        </h1>
        
        <p className="text-gray-600 text-center mb-6">
          This is a beautiful dashboard built with React and Tailwind CSS
        </p>

        <div className="bg-gradient-to-r from-blue-500 to-purple-600 rounded-xl p-6 mb-6">
          <h2 className="text-2xl font-semibold text-white mb-2">Counter</h2>
          <p className="text-6xl font-bold text-white text-center my-4">{count}</p>
          
          <div className="flex gap-3 justify-center">
            <button
              onClick={() => setCount(count - 1)}
              className="bg-red-500 hover:bg-red-600 text-white px-6 py-2 rounded-lg font-semibold transition-all transform hover:scale-105"
            >
              Decrease
            </button>
            <button
              onClick={() => setCount(0)}
              className="bg-gray-700 hover:bg-gray-800 text-white px-6 py-2 rounded-lg font-semibold transition-all transform hover:scale-105"
            >
              Reset
            </button>
            <button
              onClick={() => setCount(count + 1)}
              className="bg-green-500 hover:bg-green-600 text-white px-6 py-2 rounded-lg font-semibold transition-all transform hover:scale-105"
            >
              Increase
            </button>
          </div>
        </div>

        <div className="border-t-2 border-gray-200 pt-6">
          <div className="flex items-center justify-between mb-4">
            <span className="text-lg font-semibold text-gray-700">Status</span>
            <button
              onClick={() => setIsActive(!isActive)}
              className={`px-6 py-2 rounded-full font-semibold transition-all transform hover:scale-105 ${
                isActive
                  ? 'bg-green-500 text-white hover:bg-green-600'
                  : 'bg-gray-300 text-gray-700 hover:bg-gray-400'
              }`}
            >
              {isActive ? 'Active ✓' : 'Inactive'}
            </button>
          </div>

          {isActive && (
            <div className="bg-green-100 border-l-4 border-green-500 p-4 rounded animate-pulse">
              <p className="text-green-700 font-semibold">
                🎉 System is now active and running!
              </p>
            </div>
          )}
        </div>

        <div className="mt-6 text-center">
          <p className="text-sm text-gray-500">
            Built with ❤️ using React + Vite + Tailwind CSS
          </p>
        </div>
      </div>
    </div>
  )
}

export default DashboardComponent
